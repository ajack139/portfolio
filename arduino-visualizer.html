Code snippet

Example of reading microphone input and mapping it to output.


// --- Baseline calibration (run once at startup) ---
if (!baselineSet) {
  int reading = analogRead(MIC_PIN);
  calibSum += reading;
  calibCount++;

  if (calibCount >= 500) {
    baseline = (float)calibSum / (float)calibCount;
    baselineSet = true;
  }
  delay(1);
  return;
}

// --- Read mic, center around baseline, smooth noise ---
long sum = 0;
for (int i = 0; i < 16; i++) sum += analogRead(MIC_PIN);

float raw = sum / 16.0;
float centered = fabs(raw - baseline);

// low-pass filter to reduce buzz/jitter
filteredCentered = 0.7 * filteredCentered + 0.3 * centered;
centered = filteredCentered;

// --- Convert to a stable 0–8 bar height ---
float levelAbove = (centered > QUIET_THRESHOLD) ? (centered - QUIET_THRESHOLD) : 0;

if (levelAbove <= 0) smoothedLevel *= 0.35;          // decay
else smoothedLevel = 0.3 * smoothedLevel + 0.7 * levelAbove; // attack

float fractionBar = constrain(smoothedLevel / MAX_LEVEL_DIFF, 0.0, 1.0);
int barHeight = (int)(fractionBar * 8.0 + 0.5);
